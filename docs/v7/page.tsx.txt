export default function Home() {
  const main = useRef<HTMLElement>(null);
  const cursor = useRef<HTMLDivElement>(null);
  const hold = useRef<HTMLButtonElement>(null);
  const holdTween = useRef<gsap.core.Tween | null>(null);
  const [now, setNow] = useState<Date | null>(null);
  const [join, setJoin] = useState<(typeof plans)[number] | null>(null);
  const [memberId, setMemberId] = useState("");
  const [toast, setToast] = useState(-1);

  useEffect(() => {
    const update = () => setNow(new Date());
    const t = setTimeout(update);
    const id = setInterval(update, 30000);
    return () => {
      clearTimeout(t);
      clearInterval(id);
    };
  }, []);

  // One activity popup every ~12s, visible for 5s.
  useEffect(() => {
    let i = 0;
    let hide: ReturnType<typeof setTimeout>;
    const show = () => {
      setToast(i++ % activity.length);
      hide = setTimeout(() => setToast(-1), 5000);
    };
    const first = setTimeout(show, 9000);
    const id = setInterval(show, 12000);
    return () => {
      clearTimeout(first);
      clearTimeout(hide);
      clearInterval(id);
    };
  }, []);

  const ist = now ? new Date(now.toLocaleString("en-US", { timeZone: "Asia/Kolkata" })) : null;
  const hour = ist?.getHours() ?? 18;
  const open = hour >= 5 && hour < 23;

  // Joining fee waived until the end of this month (IST).
  const deadline = ist ? new Date(ist.getFullYear(), ist.getMonth() + 1, 0, 23, 59) : null;
  const left = deadline && ist ? deadline.getTime() - ist.getTime() : 0;
  const leftText = `${Math.floor(left / 864e5)}d ${Math.floor((left / 36e5) % 24)}h ${Math.floor((left / 6e4) % 60)}m`;
  const monthEnd = deadline?.toLocaleDateString("en-IN", { day: "numeric", month: "short" });

  const openPlan = (name: string) => {
    setMemberId("");
    setJoin(plans.find((p) => p.name === name) ?? plans[1]);
  };

  // "Hold to start": press and hold like a deadlift grip.
  const startHold = () => {
    holdTween.current?.kill();
    holdTween.current = gsap.to(hold.current, {
      "--p": 1,
      duration: 1.4,
      ease: "power1.in",
      onComplete: () => {
        navigator.vibrate?.(40);
        gsap.set(hold.current, { "--p": 0 });
        openPlan("Quarterly");
      },
    });
  };
  const stopHold = () => {
    holdTween.current?.kill();
    gsap.to(hold.current, { "--p": 0, duration: 0.4, ease: "power2.out" });
  };

  useGSAP(
    () => {
      const lenis = new Lenis();
      lenis.on("scroll", ScrollTrigger.update);
      const tick = (t: number) => lenis.raf(t * 1000);
      gsap.ticker.add(tick);
      gsap.ticker.lagSmoothing(0);

      const mm = gsap.matchMedia();

      mm.add("(prefers-reduced-motion: reduce)", () => {
        gsap.set(".loader, .flash, .curtain", { autoAlpha: 0 });
        gsap.set(".nav, .mobile-bar, .side-stack, .hero-content", { autoAlpha: 1 });
      });

      mm.add({ desktop: "(min-width: 900px)", motion: "(prefers-reduced-motion: no-preference)" }, (ctx) => {
        const { desktop, motion } = ctx.conditions!;
        if (!motion) return;

        // ---- Hero shrinks into a framed card while the name splits apart.
        // Built after the opening so the two don't fight.
        const heroScroll = () =>
          gsap
            .timeline({
              scrollTrigger: {
                trigger: ".hero",
                start: "top top",
                end: "+=100%",
                pin: true,
                scrub: 0.6,
                refreshPriority: 2,
              },
            })
            .fromTo(
              ".hero-media",
              { clipPath: "inset(0% 0% 0% 0%)" },
              {
                clipPath: desktop ? "inset(16% 24% 16% 24%)" : "inset(22% 6% 22% 6%)",
                ease: "none",
                immediateRender: false,
              },
            )
            .to(".hero-media video", { scale: 1.15, ease: "none" }, 0)
            .to(".hero-word.top", { xPercent: -8, ease: "none" }, 0)
            .to(".hero-word.bottom", { xPercent: 8, ease: "none" }, 0)
            .fromTo(".hero-meta", { opacity: 1 }, { opacity: 0, ease: "none", duration: 0.3, immediateRender: false }, 0)
            .fromTo(
              ".hero-mid",
              { opacity: 0, y: 30 },
              { opacity: 1, y: 0, ease: "none", duration: 0.5, immediateRender: false },
              0.4,
            );

        // ---- Opening: quick cuts flash inside a kettlebell-shaped window (his logo),
        // then the kettlebell rushes at you until the video fills the screen.
        lenis.stop();
        // The kettlebell is drawn in a 240-wide box; its bell centre is (120, 186).
        const shape = document.querySelector(".kb-shape")!;
        const kb = { size: 0 };
        const place = () =>
          shape.setAttribute(
            "transform",
            `translate(${innerWidth / 2} ${innerHeight / 2}) scale(${kb.size / 240}) translate(-120 -186)`,
          );
        place();
        const small = Math.min(innerWidth * 0.5, innerHeight * 0.38, 320);
        const big = Math.max(innerWidth, innerHeight) * 4;
        const counter = { v: 0 };
        const countEl = document.querySelector(".loader-count");
        const intro = gsap.timeline({
          onComplete: () => {
            heroScroll();
            lenis.start();
            ScrollTrigger.refresh();
          },
        });
        intro
          .to(kb, { size: small, duration: 1, ease: "back.out(1.7)", onUpdate: place }, 0.15)
          .to(
            counter,
            {
              v: 100,
              duration: 2.7,
              ease: "power2.inOut",
              onUpdate: () => {
                if (countEl) countEl.textContent = String(Math.round(counter.v)).padStart(3, "0");
              },
            },
            0,
          )
          .from(".loader-line > span", { yPercent: 110, duration: 0.9, ease: "expo.out" }, 0.3);
        gsap.utils.toArray<HTMLElement>(".flash").forEach((el, i) => intro.set(el, { autoAlpha: 1 }, 0.45 + i * 0.27));
        intro
          .set(".flash", { autoAlpha: 0 }, 2.75)
          .to(".loader-inner", { opacity: 0, y: 20, duration: 0.5, ease: "power2.in" }, 2.7)
          .to(kb, { size: big, duration: 1.3, ease: "power4.in", onUpdate: place }, 2.75)
          .set(".curtain", { autoAlpha: 0 }, 4.05)
          .set(".loader", { autoAlpha: 0 }, 4.05)
          .set(".hero-content, .nav, .mobile-bar", { autoAlpha: 1 }, 4.0)
          .from(".hero-word .ch", { yPercent: 110, duration: 1.3, ease: "expo.out", stagger: 0.035 }, 4.0)
          .from(".hero-fade", { opacity: 0, y: 16, duration: 1, ease: "power3.out", stagger: 0.08 }, 4.35)
          .from(".nav", { y: -16, opacity: 0, duration: 0.8, ease: "power3.out" }, 4.2);

        // Offer + activity cards only after the full-screen cards, so they never cover the story.
        ScrollTrigger.create({
          trigger: "#quiz",
          start: "top 80%",
          endTrigger: "html",
          end: "bottom bottom",
          onToggle: (self) => gsap.to(".side-stack", { autoAlpha: self.isActive ? 1 : 0, duration: 0.4 }),
        });

        // ---- Cards: full size the whole time; the next one simply slides over.
        const cardEls = gsap.utils.toArray<HTMLElement>(".card");
        cardEls.forEach((card) => {
          gsap.from(card.querySelectorAll(".card-word .ch"), {
            yPercent: 110,
            duration: 1.1,
            ease: "expo.out",
            stagger: 0.03,
            scrollTrigger: { trigger: card, start: "top 40%" },
          });
          gsap.from(card.querySelectorAll(".card-fade"), {
            opacity: 0,
            y: 20,
            duration: 0.9,
            ease: "power3.out",
            stagger: 0.08,
            scrollTrigger: { trigger: card, start: "top 30%" },
          });
          gsap.fromTo(
            card.querySelector(".card-media"),
            { yPercent: -8 },
            { yPercent: 0, ease: "none", scrollTrigger: { trigger: card, start: "top bottom", end: "top top", scrub: true } },
          );
        });

        gsap.from(".plan-plate", {
          rotation: -120,
          x: -80,
          opacity: 0,
          duration: 1.3,
          ease: "expo.out",
          stagger: 0.1,
          scrollTrigger: { trigger: ".plates", start: "top 80%" },
        });
        gsap.from(".end-word .ch", {
          yPercent: 110,
          duration: 1.2,
          ease: "expo.out",
          stagger: 0.04,
          scrollTrigger: { trigger: ".end", start: "top 60%" },
        });

        if (desktop) {
          const cx = gsap.quickTo(cursor.current, "x", { duration: 0.35, ease: "power3" });
          const cy = gsap.quickTo(cursor.current, "y", { duration: 0.35, ease: "power3" });
          const move = (e: MouseEvent) => {
            cx(e.clientX);
            cy(e.clientY);
          };
          const over = (e: MouseEvent) => {
            const el = (e.target as HTMLElement).closest<HTMLElement>("[data-cursor], a, button");
            const label = el?.dataset.cursor ?? "";
            cursor.current?.classList.toggle("big", !!label);
            cursor.current?.classList.toggle("link", !!el && !label);
            const text = cursor.current?.querySelector("span");
            if (text) text.textContent = label;
          };
          window.addEventListener("mousemove", move);
          window.addEventListener("mouseover", over);
          return () => {
            window.removeEventListener("mousemove", move);
            window.removeEventListener("mouseover", over);
          };
        }
      });

      return () => {
        gsap.ticker.remove(tick);
        lenis.destroy();
      };
    },
    { scope: main },
  );

  return (
    <main ref={main}>
      {/* OPENING */}
      <div className="loader" aria-hidden>
        <div className="loader-inner">
          <p className="loader-line">
            <span>The Muscle Paradise · Date the weight</span>
          </p>
          <p className="loader-count">000</p>
        </div>
      </div>

      <nav className="nav">
        <a href="#top" className="logo" aria-label="The Muscle Paradise home">
          <Logo variant="mark" />
          The Muscle Paradise
        </a>
        <div className="nav-links">
          <a href="#cardio">Floors</a>
          <a href="#stage">Stage</a>
          <a href="#quiz">Your plan</a>
          <a href="#plans">Membership</a>
        </div>
        <div className="nav-right">
          <span className="status">
            <i className={open ? "on" : ""} />
            {ist
              ? `${open ? "Open now" : "Closed"} · ${ist.toLocaleTimeString("en-IN", { hour: "numeric", minute: "2-digit" })}`
              : "Kanpur"}
          </span>
          <a href="#plans" className="btn">
            Join
          </a>
        </div>
      </nav>

      {/* HERO */}
      <section className="hero" id="top">
        <div className="hero-media">
          <video src="/video/hero.mp4" poster="/video/hero.jpg" autoPlay muted loop playsInline />
          {flashes.map((src) => (
            <Image key={src} className="flash" src={src} alt="" fill sizes="40vw" />
          ))}
        </div>
        {/* Opening curtain: a black sheet with his kettlebell cut out of it. */}
        <svg className="curtain" aria-hidden>
          <defs>
            <mask id="kb-hole">
              <rect width="100%" height="100%" fill="#fff" />
              <g className="kb-shape">
                <path
                  fill="#000"
                  fillRule="evenodd"
                  d="M28 140 V78 C28 38 56 14 96 14 H144 C184 14 212 38 212 78 V140 Z M66 120 V84 C66 60 80 48 104 48 H136 C160 48 174 60 174 84 V120 Z"
                />
                <circle cx="120" cy="186" r="108" fill="#000" />
              </g>
            </mask>
          </defs>
          <rect width="100%" height="100%" fill="#0c0c0c" mask="url(#kb-hole)" />
        </svg>
        <div className="hero-content">
          <div className="hero-meta hero-fade">
            <p>
              The Muscle Paradise
              <br />
              Unisex gym · Naubasta, Kanpur
            </p>
            <p className="right">
              Two floors · Open 5 AM – 11 PM
              <br />
              First session free
            </p>
          </div>
          <h1 className="hero-title" aria-label="The Muscle Paradise">
            {[
              ["Muscle", "top"],
              ["Paradise", "bottom"],
            ].map(([word, pos]) => (
              <span className={`hero-word ${pos}`} key={word} aria-hidden>
                {word.split("").map((c, i) => (
                  <span className="ch" key={i}>
                    {c}
                  </span>
                ))}
              </span>
            ))}
          </h1>
          <p className="hero-mid">Date the weight.</p>
          <div className="hero-bottom hero-meta">
            <p className="hero-fade">Scroll to enter</p>
            <div className="hero-fade hero-actions">
              <a className="btn big brand" href="#quiz">
                Get my plan
              </a>
              <a className="btn big ghost" href={wa("Hi! I'd like to book my free first session at The Muscle Paradise.")}>
                Book a free session
              </a>
            </div>
          </div>
        </div>
      </section>

      {/* STORY — full-screen cards that stack as you scroll */}
      <div className="stack" id="floors">
        {cards.map((c, i) => (
          <section className="card" id={c.id} key={c.id}>
            <div className="card-inner">
              <div className="card-media">
                {c.video ? (
                  <video src={c.video} poster={c.poster} autoPlay muted loop playsInline preload="metadata" />
                ) : (
                  <Image src={c.photo!} alt="" fill sizes="100vw" />
                )}
              </div>
              <div className="card-copy">
                <p className="card-top card-fade">
                  <span>{c.label}</span>
                  <span>
                    {String(i + 1).padStart(2, "0")} / {String(cards.length).padStart(2, "0")}
                  </span>
                </p>
                <h2 className="card-word" aria-label={c.word}>
                  {c.word.split("").map((ch, j) => (
                    <span className="ch" key={j} aria-hidden>
                      {ch}
                    </span>
                  ))}
                </h2>
                <div className="card-foot">
                  <p className="card-line card-fade">{c.line}</p>
                  {c.cta && (
                    <a className="btn big brand card-fade" href={wa(c.msg!)}>
                      {c.cta}
                    </a>
                  )}
                </div>
              </div>
            </div>
          </section>
        ))}
      </div>

      <Quiz
        crowd={crowd}
        prices={Object.fromEntries(plans.map((p) => [p.name, p.price]))}
        onJoin={openPlan}
        wa={wa}
      />

      {/* PLANS — plates */}
      <section className="plans" id="plans">
        <div className="section-head">
          <p className="eyebrow">Membership</p>
          <h2 className="h2">Pick your plate. Heavier saves more.</h2>
        </div>
        <div className="plates">
          {plans.map((p) => (
            <div className="plan-col" key={p.name}>
              <button
                className={`plan-plate ${p.color}`}
                style={{ width: `${60 + p.kg * 1.6}%` }}
                data-cursor="Join"
                onClick={() => openPlan(p.name)}
                aria-label={`Join ${p.name} for ${inr(p.price)}`}
              >
                <span className="pp-kg">{p.kg}</span>
                <span className="pp-hole" />
                <span className="pp-label">{p.months === 1 ? "1 month" : `${p.months} months`}</span>
              </button>
              <h3>{p.name}</h3>
              <p className="price">{inr(p.price)}</p>
              <p className="plan-note">{p.note}</p>
              <button className={`btn ${p.months === 12 ? "brand" : "ghost"}`} onClick={() => openPlan(p.name)}>
                Join {p.name.toLowerCase()}
              </button>
            </div>
          ))}
        </div>
        <p className="addons">
          Personal training ₹4,000 a month · Couples save 15% · Students save 10% with ID · GST invoice with every payment
        </p>
      </section>

      {/* END — one last full screen */}
      <footer className="end">
        <div className="end-media">
          <Image src="/photos/storefront.jpg" alt="The Muscle Paradise storefront with the kettlebell sign" fill sizes="100vw" />
        </div>
        <div className="end-copy">
          <p className="card-top">
            <span>Khadepur Chauraha · Naubasta</span>
            <span>Date the weight</span>
          </p>
          <h2 className="end-word" aria-label="Come lift">
            {"Come lift.".split("").map((ch, j) => (
              <span className="ch" key={j} aria-hidden>
                {ch === " " ? "\u00a0" : ch}
              </span>
            ))}
          </h2>
          <div className="end-row">
            <div className="end-info">
              <p>{ADDRESS}</p>
              <p>
                Mon – Sat 5 AM – 11 PM · Sun 6 AM – 12 PM
                <br />
                <a href={`tel:+${PHONE}`}>{PHONE_SHOWN}</a> ·{" "}
                <a href={INSTAGRAM} target="_blank" rel="noreferrer">
                  @the_muscleparadise
                </a>
              </p>
            </div>
            <div className="end-actions">
              <button
                ref={hold}
                className="hold"
                onPointerDown={startHold}
                onPointerUp={stopHold}
                onPointerLeave={stopHold}
                onKeyDown={(e) => e.key === "Enter" && openPlan("Quarterly")}
                onContextMenu={(e) => e.preventDefault()}
              >
                <svg viewBox="0 0 100 100" aria-hidden>
                  <circle cx="50" cy="50" r="46" />
                  <circle cx="50" cy="50" r="46" className="hold-ring" />
                </svg>
                <span>
                  Hold to
                  <br />
                  join
                </span>
              </button>
              <a className="btn big" href={wa("Hi! I'd like to visit The Muscle Paradise.")}>
                WhatsApp us
              </a>
              <a className="btn big ghost" href={MAPS} target="_blank" rel="noreferrer">
                Directions
              </a>
            </div>
          </div>
          <p className="end-base">© {ist?.getFullYear()} The Muscle Paradise, Kanpur</p>
        </div>
      </footer>

      <a className="wa-float" href={wa("Hi Muscle Paradise!")} aria-label="Chat on WhatsApp">
        <svg viewBox="0 0 24 24" aria-hidden>
          <path d="M12 2a10 10 0 0 0-8.6 15.1L2 22l5-1.3A10 10 0 1 0 12 2Zm5.3 14.1c-.2.6-1.3 1.2-1.8 1.2-.5.1-1 .2-3.4-.7-2.8-1.1-4.6-4-4.8-4.2-.1-.2-1.1-1.5-1.1-2.9s.7-2 1-2.3c.2-.3.5-.3.7-.3h.5c.2 0 .4 0 .6.5l.9 2.1c.1.2.1.3 0 .5l-.3.5-.4.5c-.1.1-.3.3-.1.6.2.3.7 1.2 1.6 1.9 1.1 1 2 1.3 2.3 1.4.3.1.5.1.6-.1l.9-1c.2-.3.4-.2.6-.1l2 1c.3.1.5.2.5.3.1.2.1.7-.1 1.4Z" />
        </svg>
      </a>

      {/* Offer + activity (bottom left) */}
      <div className="side-stack">
        {toast >= 0 && (
          <p className="toast" key={toast}>
            <i />
            {activity[toast]}
            <small>just now</small>
          </p>
        )}
        {ist && (
          <a href="#plans" className="offer">
            <b>No joining fee</b> if you join by {monthEnd}
            <span>{leftText} left</span>
          </a>
        )}
      </div>

      {/* Mobile sticky actions */}
      <div className="mobile-bar">
        <a className="btn brand" href="#quiz">
          Get my plan
        </a>
        <a className="btn" href={wa("Hi! I'd like to book my free first session at The Muscle Paradise.")}>
          Free session
        </a>
      </div>

      <div className="cursor" ref={cursor} aria-hidden>
        <span />
      </div>

      {/* JOIN SHEET */}
      {join && (
        <div className="sheet-bg" onClick={() => setJoin(null)} data-lenis-prevent>
          <div className="sheet" onClick={(e) => e.stopPropagation()} role="dialog" aria-modal aria-label="Join">
            <button className="close" onClick={() => setJoin(null)}>
              Close
            </button>
            {!memberId ? (
              <form
                onSubmit={(e) => {
                  e.preventDefault();
                  setMemberId("MP-" + Math.floor(1000 + Math.random() * 9000));
                }}
              >
                <div className={`sheet-plate ${join.color}`}>{join.kg}</div>
                <p className="eyebrow">You picked</p>
                <h3>
                  {join.name} · {inr(join.price)}
                </h3>
                <label>
                  <span>Full name</span>
                  <input required name="name" autoComplete="name" />
                </label>
                <label>
                  <span>Mobile</span>
                  <input
                    required
                    name="phone"
                    type="tel"
                    inputMode="numeric"
                    pattern="[6-9][0-9]{9}"
                    placeholder="10-digit number"
                  />
                </label>
                <label>
                  <span>Start date</span>
                  <input required name="start" type="date" />
                </label>
                <button className="btn big brand" type="submit">
                  Pay {inr(join.price)} with UPI
                </button>
                <p className="small">UPI, cards, net banking, EMI on yearly. Demo only: Razorpay connects in phase 2.</p>
              </form>
            ) : (
              <div className="done">
                <div className={`sheet-plate ${join.color} spin`}>{join.kg}</div>
                <p className="eyebrow">Payment received</p>
                <h3>You&apos;re in.</h3>
                <p>
                  Membership <b>{memberId}</b> is active. Your receipt and GST invoice are on WhatsApp. Put your finger on
                  the scanner at the door. It already knows you.
                </p>
                <button className="btn big" onClick={() => setJoin(null)}>
                  See you on the floor
                </button>
              </div>
            )}
          </div>
        </div>
      )}
    </main>
  );
}
