              window.removeEventListener("mousemove", move);
              window.removeEventListener("mouseover", over);

function splitWords(s: string) {
  return s.split(" ").map((w, i) => (
    <span className="w" key={i}>
      <span>{w}&nbsp;</span>
    </span>
  ));
}
